init = main(n)

main(n: int) = if n >= 0 then proc0(n) else unit()
proc0(n: int) = new s1, r1 in proc1(n, s1, r1)
proc1(n: int, s1: Sender, r1: Receiver) = new s2, r2 in proc2(n, s1, s2, r1, r2)
proc2(n: int, s1: Sender, s2: Sender, r1: Receiver, r2: Receiver) = spawn(proc3(s1, r2)); proc4(n, 1, r1, s2, 0)
proc3(s1: Sender, r2: Receiver) = let v = recv r2 in proc3_1(s1, v, r2)
proc3_1(s1: Sender, v: int, r2: Receiver) = send v + 1 to s1; proc3(s1, r2)
proc4(n: int, count: int, r1: Receiver, s2: Sender, sum: int) = if count < n then proc4_1(n, count, r1, s2, sum) else proc5(n, sum, r1, s2)
proc4_1(n: int, count: int, r1: Receiver, s2: Sender, sum: int) = send 1 to s2; proc4_2(n, count, r1, s2, sum)
proc4_2(n: int, count: int, r1: Receiver, s2: Sender, sum: int) = let v = recv r1 in proc4(n, count + 1, r1, s2, sum + v)
proc5(n: int, v: int, r1: Receiver, s2: Sender) = spawn(drop(s2, r1)); assert(v == n * (1 + 1))

assert(cond: int) = if cond then unit() else fail()
unit() = ()
fail() = fail

drop(s: Sender, r: Receiver) = ()
