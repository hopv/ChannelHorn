init = main()

main() = new s, r in f1(s, r)
f1(s: Sender, r: Receiver) = new p, q in f2(s, r, p, q)
f2(s: Sender, r: Receiver, p: Sender, q: Receiver) = spawn(server(s, q)); client(r, p, 0)
server(s: Sender, q: Receiver) = let x = recv q in server_inner1(s, q, x)
server_inner1(s: Sender, q: Receiver, x: int) = spawn(send_fun(s, x + 1)); drop_r(q)
client(r: Receiver, p: Sender, x: int) = spawn(send_fun(p, x)); client_inner2(r, x)
client_inner2(r: Receiver, x: int) = let y = recv r in client_inner3(r, x, y)
client_inner3(r: Receiver, x: int, y: int) = spawn(assert(y == x + 1)); drop_r(r)

assert(cond: int) = if cond then unit() else fail()
unit() = ()
fail() = fail

drop(s: Sender, r: Receiver) = ()
drop_s(s: Sender) = ()
drop_r(r: Receiver) = ()

send_fun(s: Sender, val: int) = send val to s; drop_s(s)
