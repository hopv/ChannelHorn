init = main()

main() = new s1, r1 in proc1(s1, r1)
proc1(s1: Sender, r1: Receiver) = new s2, r2 in proc2(s1, s2, r1, r2)
proc2(s1: Sender, s2: Sender, r1: Receiver, r2: Receiver) = spawn(proc3(s1, r2)); proc4(r1, s2)
proc3(s1: Sender, r2: Receiver) = let v = recv r2 in proc3_1(s1, v, r2)
proc3_1(s1: Sender, v: int, r2: Receiver) = send v + 1 to s1; proc3(s1, r2)
proc4(r1: Receiver, s2: Sender) = send 0 to s2; proc5(r1, s2)
proc5(r1: Receiver, s2: Sender) = let v = recv r1 in proc6(r1, s2, v)
proc6(r1: Receiver, s2: Sender, v: int) = send v + 1 to s2; proc7(r1, s2)
proc7(r1: Receiver, s2: Sender) = let v = recv r1 in proc8(v, r1, s2)
proc8(v: int, r1: Receiver, s2: Sender) = spawn(drop(s2, r1)); assert(v == 3)

assert(cond: int) = if cond then unit() else fail()
unit() = ()
fail() = fail

drop(s: Sender, r: Receiver) = ()
